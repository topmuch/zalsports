   171→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   172→  const slots: TimeSlot[] = [];
   173→  for (let h = 8; h <= 23; h++) {
   174→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   175→    const available = availableHours ? !availableHours.has(timeStr) : true;
   176→    slots.push({
   177→      id: `slot-${h}`,
   178→      time: timeStr,
   179→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   180→      available,
   181→      price: 25000,
   182→    });
   183→  }
   184→  return slots;
   185→}

   187→/* ═══════════════════════════════════════════
   188→   Animated Counter
   189→   ═══════════════════════════════════════════ */
   190→
   191→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   192→  const [count, setCount] = useState(0);
   193→  useEffect(() => {
   194→    if (!inView) return;
   195→    let start = 0;
   196→    const step = Math.max(1, Math.floor(value / 125));
   197→    const timer = setInterval(() => {
   198→      start += step;
   199→      if (start >= value) {
   200→        setCount(value);
   201→        clearInterval(timer);
   202→      } else {
   203→        setCount(start);
   204→      }
   205→    }, 16);
   206→    return () => clearInterval(timer);
   207→  }, [inView, value]);
   208→
   209→  return (
   210→    <span>
   211→      {count.toLocaleString('fr-FR')}
   212→      {suffix}
   213→    </span>
   214→  );
   215→}

   220→
   221→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   222→  const ref = useRef<HTMLDivElement>(null);
   223→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   224→  return (
   225→    <motion.section
   226→      ref={ref}
   227→      id={id}
   228→      className={className}
   229→      initial={{ opacity: 0, y: 30 }}
   230→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   231→      transition={{ duration: 0.6, ease: 'easeOut' }}
   232→    >
   233→      {children}
   234→    </motion.section>
   235→  );
   236→}

   278→  const [phone, setPhone] = useState('');
   279→  const [isSubmitting, setIsSubmitting] = useState(false);
   280→  const [loadingSlots, setLoadingSlots] = useState(false);
   281→  const [paymentMethod, setPaymentMethod] = useState<'wave' | 'orange_money' | 'cash' | null>(null);
   282→  const [paymentPhone, setPaymentPhone] = useState('');
   283→  const [bookingId, setBookingId] = useState<string | null>(null);
   284→  const [isPaying, setIsPaying] = useState(false);
   285→  const [fullyBookedDates, setFullyBookedDates] = useState<string[]>([]);
   286→
   287→  const fullyBookedSet = useMemo(() => new Set(fullyBookedDates), [fullyBookedDates]);
   288→
   289→  const fetchSlots = useCallback(async (date: Date) => {
   290→    setLoadingSlots(true);
   291→    try {
   292→      const dateStr = format(date, 'yyyy-MM-dd');
   293→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   294→      if (res.ok) {
   295→        const data = await res.json();
   296→        const bookedSet = new Set(
   297→          data.available
   298→            .filter((s: { time: string; available: boolean }) => !s.available)
   299→            .map((s: { time: string; available: boolean }) => s.time)
   300→        );
   301→        setSlots(generateTimeSlots(bookedSet));
   302→      } else {
   303→        setSlots(generateTimeSlots());
   304→      }
   305→    } catch {
   306→      setSlots(generateTimeSlots());
   307→    } finally {
   308→      setLoadingSlots(false);
   309→    }
   310→  }, []);
   311→
   312→  useEffect(() => {
   313→    if (open) {
   314→      const tomorrow = addDays(new Date(), 1);
   315→      setSelectedDate(tomorrow);
   316→      setSelectedSlot(null);
   317→      setStep('select');
   318→      setName('');
   319→      setPhone('');
   320→      setPaymentMethod(null);
   321→      setPaymentPhone('');
   322→      setBookingId(null);
   323→      setIsPaying(false);
   324→      fetchSlots(tomorrow);
   325→    }
   326→  }, [open, fetchSlots]);
   327→
   328→  useEffect(() => {
   329→    if (open && selectedDate) {
   330→      setSelectedSlot(null);
   331→      fetchSlots(selectedDate);
   332→    }
   333→  }, [selectedDate, open, fetchSlots]);
   334→
   335→  // Fetch fully booked dates for calendar coloring
   336→  useEffect(() => {
   337→    if (!open || !selectedDate) return;
   338→    const monthStr = format(selectedDate, 'yyyy-MM');
   339→    fetch(`/api/bookings/month-availability?month=${monthStr}`)
   340→      .then((res) => res.ok ? res.json() : null)
   341→      .then((data) => {
   342→        if (data?.fullyBooked) setFullyBookedDates(data.fullyBooked);
   343→      })
   344→      .catch(() => {});
   345→  }, [open, selectedDate]);
   346→
   347→  const handleConfirmInfo = useCallback(async () => {
   348→    if (!selectedSlot || !selectedDate || !name || !phone) return;
   349→    setIsSubmitting(true);
   350→    try {
   351→      const res = await fetch('/api/bookings', {
   352→        method: 'POST',
   353→        headers: { 'Content-Type': 'application/json' },
   354→        body: JSON.stringify({
   355→          date: format(selectedDate, 'yyyy-MM-dd'),
   356→          timeSlot: selectedSlot.time,
   357→          customerName: name,
   358→          customerPhone: phone,
   359→        }),
   360→      });
   361→      if (res.ok) {
   362→        const data = await res.json();
   363→        setBookingId(data.id || data.booking?.id || null);
   364→        setPaymentPhone(phone);
   365→        setStep('payment');
   366→      }
   367→    } catch {
   368→      // silent
   369→    } finally {
   370→      setIsSubmitting(false);
   371→    }
   372→  }, [selectedSlot, selectedDate, name, phone]);
   373→
   374→  const handleInitiatePayment = useCallback(async () => {
   375→    if (!paymentMethod || !bookingId) return;
   376→    if (paymentMethod !== 'cash' && !paymentPhone) return;
   377→
   378→    setIsPaying(true);
   379→    try {
   380→      // For all methods: directly confirm the booking with payment info
   381→      // In production, Wave/OM would redirect to a real payment gateway
   382→      await fetch(`/api/bookings/${bookingId}`, {
   383→        method: 'POST',
   384→        headers: { 'Content-Type': 'application/json' },
   385→        body: JSON.stringify({
   386→          status: 'confirmed',
   387→          paymentStatus: paymentMethod === 'cash' ? 'pending' : 'partial',
   388→          paymentMethod,
   389→          depositPaid: paymentMethod === 'cash' ? 0 : 5000,
   390→        }),
   391→      });
   392→      setStep('confirm');
   393→    } catch {
   394→      // silent
   395→    } finally {
   396→      setIsPaying(false);
   397→    }
   398→  }, [paymentMethod, paymentPhone, bookingId]);
   399→
   400→  return (
   401→    <Dialog open={open} onOpenChange={onOpenChange}>
   402→      <DialogContent className="sm:max-w-4xl md:max-w-5xl bg-white border-border max-h-[95vh] overflow-hidden flex flex-col p-0">
   403→        {step === 'confirm' ? (
