    95→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    96→  const slots: TimeSlot[] = [];
    97→  for (let h = 8; h <= 23; h++) {
    98→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    99→    const available = availableHours ? !availableHours.has(timeStr) : true;
   100→    slots.push({
   101→      id: `slot-${h}`,
   102→      time: timeStr,
   103→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   104→      available,
   105→      price: 25000,
   106→    });
   107→  }
   108→  return slots;
   109→}

   111→/* ───────────── Animated Counter ───────────── */
   112→function AnimatedCounter({
   113→  value,
   114→  suffix,
   115→  inView,
   116→}: {
   117→  value: number;
   118→  suffix: string;
   119→  inView: boolean;
   120→}) {
   121→  const [count, setCount] = useState(0);
   122→
   123→  useEffect(() => {
   124→    if (!inView) return;
   125→    let start = 0;
   126→    const duration = 2000;
   127→    const step = Math.max(1, Math.floor(value / (duration / 16)));
   128→    const timer = setInterval(() => {
   129→      start += step;
   130→      if (start >= value) {
   131→        setCount(value);
   132→        clearInterval(timer);
   133→      } else {
   134→        setCount(start);
   135→      }
   136→    }, 16);
   137→    return () => clearInterval(timer);
   138→  }, [inView, value]);
   139→
   140→  return (
   141→    <span>
   142→      {count.toLocaleString('fr-FR')}
   143→      {suffix}
   144→    </span>
   145→  );
   146→}

   152→  id,
   153→}: {

   155→  className?: string;

   161→  return (
   162→    <motion.section
   163→      ref={ref}
   164→      id={id}
   165→      className={className}
   166→      initial={{ opacity: 0, y: 40 }}
   167→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   168→      transition={{ duration: 0.7, ease: 'easeOut' }}
   169→    >
   170→      {children}
   171→    </motion.section>
   172→  );
   173→}

   184→    addDays(new Date(), 1)

   187→  const [slots, setSlots] = useState<TimeSlot[]>([]);
   188→  const [step, setStep] = useState<'select' | 'info' | 'confirm'>('select');
   189→  const [name, setName] = useState('');
   190→  const [phone, setPhone] = useState('');
   191→  const [isSubmitting, setIsSubmitting] = useState(false);
   192→
   193→  const [loadingSlots, setLoadingSlots] = useState(false);
   194→
   195→  const fetchSlots = useCallback(async (date: Date) => {
   196→    setLoadingSlots(true);
   197→    try {
   198→      const dateStr = format(date, 'yyyy-MM-dd');
   199→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   200→      if (res.ok) {
   201→        const data = await res.json();
   202→        const bookedSet = new Set(
   203→          data.available
   204→            .filter((s: { time: string; available: boolean }) => !s.available)
   205→            .map((s: { time: string; available: boolean }) => s.time)
   206→        );
   207→        setSlots(generateTimeSlots(bookedSet));
   208→      } else {
   209→        setSlots(generateTimeSlots());
   210→      }
   211→    } catch {
   212→      setSlots(generateTimeSlots());
   213→    } finally {
   214→      setLoadingSlots(false);
   215→    }
   216→  }, []);
   217→
   218→  useEffect(() => {
   219→    if (open) {
   220→      const tomorrow = addDays(new Date(), 1);
   221→      setSelectedDate(tomorrow);
   222→      setSelectedSlot(null);
   223→      setStep('select');
   224→      setName('');
   225→      setPhone('');
   226→      fetchSlots(tomorrow);
   227→    }
   228→  }, [open, fetchSlots]);
   229→
   230→  // Refetch slots when date changes
   231→  useEffect(() => {
   232→    if (open && selectedDate) {
   233→      setSelectedSlot(null);
   234→      fetchSlots(selectedDate);
   235→    }

   238→  const availableSlots = selectedDate ? slots : [];

   242→    setIsSubmitting(true);

   244→      const res = await fetch('/api/bookings', {
   245→        method: 'POST',
   246→        headers: { 'Content-Type': 'application/json' },

   248→          date: format(selectedDate, 'yyyy-MM-dd'),
   249→          timeSlot: selectedSlot.time,
   250→          customerName: name,
   251→          customerPhone: phone,
   252→        }),
   253→      });

   255→        setStep('confirm');
   256→      }
   257→    } catch {
   258→      // silent
   259→    } finally {
   260→      setIsSubmitting(false);
   261→    }
   262→  }, [selectedSlot, selectedDate, name, phone]);
   263→
   264→  return (
   265→    <Dialog open={open} onOpenChange={onOpenChange}>
   266→      <DialogContent className="sm:max-w-2xl bg-card border-border max-h-[90vh] overflow-hidden flex flex-col p-0">
   267→        {step === 'confirm' ? (

   271→            </div>
   272→            <DialogHeader>
   273→              <DialogTitle className="text-2xl">
   274→                Réservation confirmée !
   275→              </DialogTitle>
   276→              <DialogDescription className="text-muted-foreground text-base mt-2">
   277→                Votre créneau a été réservé avec succès. Vous recevrez un
   278→                SMS de confirmation.
   279→              </DialogDescription>

   281→            <div className="bg-secondary rounded-xl p-4 mt-4 w-full max-w-sm space-y-2">
   282→              <div className="flex justify-between text-sm">
   283→                <span className="text-muted-foreground">Date</span>
   284→                <span className="font-medium">
   285→                  {selectedDate &&
   286→                    format(selectedDate, "EEEE d MMMM yyyy", { locale: fr })}
   287→                </span>
   288→              </div>
   289→              <div className="flex justify-between text-sm">
   290→                <span className="text-muted-foreground">Créneau</span>
   291→                <span className="font-medium">{selectedSlot?.label}</span>
