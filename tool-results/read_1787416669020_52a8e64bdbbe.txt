    97→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    98→  const slots: TimeSlot[] = [];
    99→  for (let h = 8; h <= 23; h++) {
   100→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   101→    const available = availableHours ? !availableHours.has(timeStr) : true;
   102→    slots.push({
   103→      id: `slot-${h}`,
   104→      time: timeStr,
   105→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   106→      available,
   107→      price: 25000,
   108→    });
   109→  }
   110→  return slots;
   111→}

   113→/* ───────────── Animated Counter ───────────── */
   114→function AnimatedCounter({
   115→  value,
   116→  suffix,
   117→  inView,
   118→}: {
   119→  value: number;
   120→  suffix: string;
   121→  inView: boolean;
   122→}) {
   123→  const [count, setCount] = useState(0);
   124→
   125→  useEffect(() => {
   126→    if (!inView) return;
   127→    let start = 0;
   128→    const duration = 2000;
   129→    const step = Math.max(1, Math.floor(value / (duration / 16)));
   130→    const timer = setInterval(() => {
   131→      start += step;
   132→      if (start >= value) {
   133→        setCount(value);
   134→        clearInterval(timer);
   135→      } else {
   136→        setCount(start);
   137→      }
   138→    }, 16);
   139→    return () => clearInterval(timer);
   140→  }, [inView, value]);
   141→
   142→  return (
   143→    <span>
   144→      {count.toLocaleString('fr-FR')}
   145→      {suffix}
   146→    </span>
   147→  );
   148→}

   190→  const [step, setStep] = useState<'select' | 'info' | 'confirm'>('select');

   198→    setLoadingSlots(true);
   199→    try {
   200→      const dateStr = format(date, 'yyyy-MM-dd');
   201→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   202→      if (res.ok) {
   203→        const data = await res.json();
   204→        const bookedSet = new Set(
   205→          data.available
   206→            .filter((s: { time: string; available: boolean }) => !s.available)
   207→            .map((s: { time: string; available: boolean }) => s.time)
   208→        );
   209→        setSlots(generateTimeSlots(bookedSet));
   210→      } else {
   211→        setSlots(generateTimeSlots());
   212→      }
   213→    } catch {
   214→      setSlots(generateTimeSlots());
   215→    } finally {
   216→      setLoadingSlots(false);
   217→    }
   218→  }, []);
   219→
   220→  useEffect(() => {
   221→    if (open) {
   222→      const tomorrow = addDays(new Date(), 1);
   223→      setSelectedDate(tomorrow);
   224→      setSelectedSlot(null);
   225→      setStep('select');
   226→      setName('');
   227→      setPhone('');
   228→      fetchSlots(tomorrow);
   229→    }
   230→  }, [open, fetchSlots]);
   231→

   233→  useEffect(() => {

   235→      setSelectedSlot(null);

   253→          customerPhone: phone,
   254→        }),

   257→        setStep('confirm');
   258→      }
   259→    } catch {
   260→      // silent
   261→    } finally {
   262→      setIsSubmitting(false);
   263→    }
   264→  }, [selectedSlot, selectedDate, name, phone]);

   266→  return (

   279→                Votre créneau a été réservé avec succès. Vous recevrez un

   285→                <span className="text-muted-foreground">Date</span>
   286→                <span className="font-medium">
   287→                  {selectedDate &&
   288→                    format(selectedDate, "EEEE d MMMM yyyy", { locale: fr })}
   289→                </span>
   290→              </div>
   291→              <div className="flex justify-between text-sm">
   292→                <span className="text-muted-foreground">Créneau</span>
