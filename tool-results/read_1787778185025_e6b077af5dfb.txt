   175→
   176→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   177→  const slots: TimeSlot[] = [];
   178→  for (let h = 8; h <= 23; h++) {
   179→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   180→    const available = availableHours ? !availableHours.has(timeStr) : true;
   181→    slots.push({
   182→      id: `slot-${h}`,
   183→      time: timeStr,
   184→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   185→      available,
   186→      price: 25000,
   187→    });
   188→  }
   189→  return slots;
   190→}

   192→/* ═══════════════════════════════════════════
   193→   Animated Counter
   194→   ═══════════════════════════════════════════ */
   195→
   196→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   197→  const [count, setCount] = useState(0);
   198→  useEffect(() => {
   199→    if (!inView) return;
   200→    let start = 0;
   201→    const step = Math.max(1, Math.floor(value / 125));
   202→    const timer = setInterval(() => {
   203→      start += step;
   204→      if (start >= value) {
   205→        setCount(value);
   206→        clearInterval(timer);
   207→      } else {
   208→        setCount(start);
   209→      }
   210→    }, 16);
   211→    return () => clearInterval(timer);
   212→  }, [inView, value]);
   213→
   214→  return (
   215→    <span>
   216→      {count.toLocaleString('fr-FR')}
   217→      {suffix}
   218→    </span>
   219→  );
   220→}

   225→
   226→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   227→  const ref = useRef<HTMLDivElement>(null);
   228→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   229→  return (
   230→    <motion.section
   231→      ref={ref}
   232→      id={id}
   233→      className={className}
   234→      initial={{ opacity: 0, y: 30 }}
   235→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   236→      transition={{ duration: 0.6, ease: 'easeOut' }}
   237→    >
   238→      {children}
   239→    </motion.section>
   240→  );
   241→}

   252→  const [name, setName] = useState('');
   253→  const [phone, setPhone] = useState('');
   254→  const [isSubmitting, setIsSubmitting] = useState(false);
   255→  const [loadingSlots, setLoadingSlots] = useState(false);
   256→  const [paymentMethod, setPaymentMethod] = useState<'wave' | 'orange_money' | 'cash' | null>(null);
   257→  const [paymentPhone, setPaymentPhone] = useState('');

   259→  const [isPaying, setIsPaying] = useState(false);
   260→  const [confirmingPayment, setConfirmingPayment] = useState(false);

   263→    setLoadingSlots(true);
   264→    try {
   265→      const dateStr = format(date, 'yyyy-MM-dd');
   266→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   267→      if (res.ok) {
   268→        const data = await res.json();
   269→        const bookedSet = new Set(
   270→          data.available
   271→            .filter((s: { time: string; available: boolean }) => !s.available)
   272→            .map((s: { time: string; available: boolean }) => s.time)
   273→        );
   274→        setSlots(generateTimeSlots(bookedSet));
   275→      } else {
   276→        setSlots(generateTimeSlots());
   277→      }
   278→    } catch {
   279→      setSlots(generateTimeSlots());
   280→    } finally {
   281→      setLoadingSlots(false);
   282→    }
   283→  }, []);
   284→
   285→  useEffect(() => {
   286→    if (open) {
   287→      const tomorrow = addDays(new Date(), 1);
   288→      setSelectedDate(tomorrow);
   289→      setSelectedSlot(null);
   290→      setStep('select');
   291→      setName('');
   292→      setPhone('');
   293→      setPaymentMethod(null);
   294→      setPaymentPhone('');
   295→      setBookingId(null);
   296→      setIsPaying(false);
   297→      setConfirmingPayment(false);
   298→      fetchSlots(tomorrow);
   299→    }
   300→  }, [open, fetchSlots]);
   301→
   302→  useEffect(() => {

   306→    }
   307→  }, [selectedDate, open, fetchSlots]);
   308→

   311→    setIsSubmitting(true);
   312→    try {
   313→      const res = await fetch('/api/bookings', {
   314→        method: 'POST',
   315→        headers: { 'Content-Type': 'application/json' },
   316→        body: JSON.stringify({
   317→          date: format(selectedDate, 'yyyy-MM-dd'),
   318→          timeSlot: selectedSlot.time,
   319→          customerName: name,
   320→          customerPhone: phone,
   321→        }),
   322→      });
   323→      if (res.ok) {
   324→        const data = await res.json();
   325→        setBookingId(data.id || data.booking?.id || null);
   326→        setPaymentPhone(phone);
   327→        setStep('payment');
   328→      }
   329→    } catch {
   330→      // silent
   331→    } finally {
   332→      setIsSubmitting(false);
   333→    }
   334→  }, [selectedSlot, selectedDate, name, phone]);
   335→
   336→  const handleInitiatePayment = useCallback(async () => {
   337→    if (!paymentMethod || !bookingId) return;
   338→    if (paymentMethod !== 'cash' && !paymentPhone) return;
   339→
   340→    // Cash: skip payment API, confirm directly
   341→    if (paymentMethod === 'cash') {
   342→      await fetch(`/api/bookings/${bookingId}`, {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({
   346→          paymentStatus: 'pending',
   347→          paymentMethod: 'cash',
   348→          depositPaid: 0,
   349→        }),
   350→      });

   353→    }

   358→        method: 'POST',

   361→          bookingId,

   373→      }
