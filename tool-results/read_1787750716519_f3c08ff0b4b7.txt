   111→   111→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   112→   112→  const slots: TimeSlot[] = [];
   113→   113→  for (let h = 8; h <= 23; h++) {
   114→   114→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   115→   115→    const available = availableHours ? !availableHours.has(timeStr) : true;
   116→   116→    slots.push({
   117→   117→      id: `slot-${h}`,
   118→   118→      time: timeStr,
   119→   119→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   120→   120→      available,
   121→   121→      price: 25000,
   122→   122→    });
   123→   123→  }
   124→   124→  return slots;
   125→   125→}

   130→   130→
   131→   131→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   132→   132→  const [count, setCount] = useState(0);
   133→   133→  useEffect(() => {
   134→   134→    if (!inView) return;
   135→   135→    let start = 0;
   136→   136→    const step = Math.max(1, Math.floor(value / 125));
   137→   137→    const timer = setInterval(() => {
   138→   138→      start += step;
   139→   139→      if (start >= value) {
   140→   140→        setCount(value);
   141→   141→        clearInterval(timer);
   142→   142→      } else {
   143→   143→        setCount(start);
   144→   144→      }
   145→   145→    }, 16);
   146→   146→    return () => clearInterval(timer);
   147→   147→  }, [inView, value]);
   148→   148→
   149→   149→  return (
   150→   150→    <span>
   151→   151→      {count.toLocaleString('fr-FR')}
   152→   152→      {suffix}
   153→   153→    </span>
   154→   154→  );
   155→   155→}

   160→   160→
   161→   161→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   162→   162→  const ref = useRef<HTMLDivElement>(null);
   163→   163→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   164→   164→  return (
   165→   165→    <motion.section
   166→   166→      ref={ref}
   167→   167→      id={id}
   168→   168→      className={className}
   169→   169→      initial={{ opacity: 0, y: 30 }}
   170→   170→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   171→   171→      transition={{ duration: 0.6, ease: 'easeOut' }}
   172→   172→    >
   173→   173→      {children}
   174→   174→    </motion.section>
   175→   175→  );
   176→   176→}
