    97→    97→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    98→    98→  const slots: TimeSlot[] = [];
    99→    99→  for (let h = 8; h <= 23; h++) {
   100→   100→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   101→   101→    const available = availableHours ? !availableHours.has(timeStr) : true;
   102→   102→    slots.push({
   103→   103→      id: `slot-${h}`,
   104→   104→      time: timeStr,
   105→   105→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   106→   106→      available,
   107→   107→      price: 25000,
   108→   108→    });
   109→   109→  }
   110→   110→  return slots;

   113→   113→/* ───────────── Animated Counter ───────────── */
   114→   114→function AnimatedCounter({
   115→   115→  value,
   116→   116→  suffix,
   117→   117→  inView,
   118→   118→}: {
   119→   119→  value: number;
   120→   120→  suffix: string;
   121→   121→  inView: boolean;
   122→   122→}) {
   123→   123→  const [count, setCount] = useState(0);
   124→   124→
   125→   125→  useEffect(() => {
   126→   126→    if (!inView) return;
   127→   127→    let start = 0;
   128→   128→    const duration = 2000;
   129→   129→    const step = Math.max(1, Math.floor(value / (duration / 16)));
   130→   130→    const timer = setInterval(() => {
   131→   131→      start += step;
   132→   132→      if (start >= value) {
   133→   133→        setCount(value);
   134→   134→        clearInterval(timer);
   135→   135→      } else {
   136→   136→        setCount(start);
   137→   137→      }
   138→   138→    }, 16);
   139→   139→    return () => clearInterval(timer);
   140→   140→  }, [inView, value]);
   141→   141→
   142→   142→  return (
   143→   143→    <span>
   144→   144→      {count.toLocaleString('fr-FR')}
   145→   145→      {suffix}
   146→   146→    </span>
   147→   147→  );
   148→   148→}
