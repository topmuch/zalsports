   164→   164→
   165→   165→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   166→   166→  const [count, setCount] = useState(0);
   167→   167→  useEffect(() => {
   168→   168→    if (!inView) return;
   169→   169→    let start = 0;
   170→   170→    const step = Math.max(1, Math.floor(value / 125));
   171→   171→    const timer = setInterval(() => {
   172→   172→      start += step;
   173→   173→      if (start >= value) {
   174→   174→        setCount(value);
   175→   175→        clearInterval(timer);
   176→   176→      } else {
   177→   177→        setCount(start);
   178→   178→      }
   179→   179→    }, 16);
   180→   180→    return () => clearInterval(timer);
   181→   181→  }, [inView, value]);
   182→   182→
   183→   183→  return (
   184→   184→    <span>
   185→   185→      {count.toLocaleString('fr-FR')}
   186→   186→      {suffix}
   187→   187→    </span>
   188→   188→  );
   189→   189→}

   194→   194→
   195→   195→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   196→   196→  const ref = useRef<HTMLDivElement>(null);
   197→   197→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   198→   198→  return (
   199→   199→    <motion.section
   200→   200→      ref={ref}
   201→   201→      id={id}
   202→   202→      className={className}
   203→   203→      initial={{ opacity: 0, y: 30 }}
   204→   204→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   205→   205→      transition={{ duration: 0.6, ease: 'easeOut' }}
   206→   206→    >
   207→   207→      {children}
   208→   208→    </motion.section>
   209→   209→  );
   210→   210→}
