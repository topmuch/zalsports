    94→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    95→  const slots: TimeSlot[] = [];
    96→  for (let h = 8; h <= 23; h++) {
    97→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    98→    const available = availableHours ? !availableHours.has(timeStr) : true;
    99→    slots.push({
   100→      id: `slot-${h}`,
   101→      time: timeStr,
   102→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   103→      available,
   104→      price: 25000,
   105→    });
   106→  }
   107→  return slots;
   108→}

   110→/* ═══════════════════════════════════════════
   111→   Animated Counter

   113→
   114→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   115→  const [count, setCount] = useState(0);
   116→  useEffect(() => {
   117→    if (!inView) return;
   118→    let start = 0;
   119→    const step = Math.max(1, Math.floor(value / 125));
   120→    const timer = setInterval(() => {
   121→      start += step;
   122→      if (start >= value) {
   123→        setCount(value);
   124→        clearInterval(timer);
   125→      } else {
   126→        setCount(start);
   127→      }
   128→    }, 16);
   129→    return () => clearInterval(timer);
   130→  }, [inView, value]);
   131→
   132→  return (
   133→    <span>
   134→      {count.toLocaleString('fr-FR')}
   135→      {suffix}
   136→    </span>
   137→  );
   138→}

   144→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   145→  const ref = useRef<HTMLDivElement>(null);
   146→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   147→  return (
   148→    <motion.section
   149→      ref={ref}
   150→      id={id}
   151→      className={className}
   152→      initial={{ opacity: 0, y: 30 }}
   153→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   154→      transition={{ duration: 0.6, ease: 'easeOut' }}
   155→    >
   156→      {children}
   157→    </motion.section>
   158→  );
   159→}

   165→function BookingDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   166→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(addDays(new Date(), 1));
   167→  const [selectedSlot, setSelectedSlot] = useState<TimeSlot | null>(null);
   168→  const [slots, setSlots] = useState<TimeSlot[]>([]);
   169→  const [step, setStep] = useState<'select' | 'info' | 'confirm'>('select');
   170→  const [name, setName] = useState('');
   171→  const [phone, setPhone] = useState('');
   172→  const [isSubmitting, setIsSubmitting] = useState(false);
   173→  const [loadingSlots, setLoadingSlots] = useState(false);
   174→
   175→  const fetchSlots = useCallback(async (date: Date) => {
   176→    setLoadingSlots(true);
   177→    try {
   178→      const dateStr = format(date, 'yyyy-MM-dd');
   179→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   180→      if (res.ok) {
   181→        const data = await res.json();
   182→        const bookedSet = new Set(
   183→          data.available
   184→            .filter((s: { time: string; available: boolean }) => !s.available)
   185→            .map((s: { time: string; available: boolean }) => s.time)
   186→        );
   187→        setSlots(generateTimeSlots(bookedSet));
   188→      } else {
   189→        setSlots(generateTimeSlots());
   190→      }
   191→    } catch {
   192→      setSlots(generateTimeSlots());
   193→    } finally {
   194→      setLoadingSlots(false);
   195→    }
   196→  }, []);
   197→
   198→  useEffect(() => {
   199→    if (open) {
   200→      const tomorrow = addDays(new Date(), 1);
   201→      setSelectedDate(tomorrow);
   202→      setSelectedSlot(null);
   203→      setStep('select');
   204→      setName('');
   205→      setPhone('');
   206→      fetchSlots(tomorrow);
   207→    }
   208→  }, [open, fetchSlots]);
   209→
   210→  useEffect(() => {
   211→    if (open && selectedDate) {
   212→      setSelectedSlot(null);
   213→      fetchSlots(selectedDate);
   214→    }
   215→  }, [selectedDate, open, fetchSlots]);

   219→    setIsSubmitting(true);

   221→      const res = await fetch('/api/bookings', {
   222→        method: 'POST',
   223→        headers: { 'Content-Type': 'application/json' },
   224→        body: JSON.stringify({
   225→          date: format(selectedDate, 'yyyy-MM-dd'),
   226→          timeSlot: selectedSlot.time,
   227→          customerName: name,
   228→          customerPhone: phone,
   229→        }),
   230→      });
   231→      if (res.ok) setStep('confirm');
   232→    } catch {
   233→      // silent
   234→    } finally {
   235→      setIsSubmitting(false);
   236→    }
   237→  }, [selectedSlot, selectedDate, name, phone]);
   238→
   239→  return (
   240→    <Dialog open={open} onOpenChange={onOpenChange}>
   241→      <DialogContent className="sm:max-w-2xl bg-card border-border max-h-[90vh] overflow-hidden flex flex-col p-0">
   242→        {step === 'confirm' ? (
   243→          <div className="flex-1 flex flex-col items-center justify-center p-8 gap-4 text-center">
   244→            <div className="w-16 h-16 rounded-full bg-primary/20 flex items-center justify-center">
   245→              <Zap className="w-8 h-8 text-primary" />
   246→            </div>
   247→            <DialogHeader>
   248→              <DialogTitle className="text-2xl">Réservation confirmée !</DialogTitle>
   249→              <DialogDescription className="text-muted-foreground text-base mt-2">
   250→                Votre créneau a été réservé avec succès. Vous recevrez un SMS de confirmation.
   251→              </DialogDescription>
   252→            </DialogHeader>
   253→            <div className="bg-secondary rounded-xl p-4 mt-4 w-full max-w-sm space-y-2">
   254→              <div className="flex justify-between text-sm">
   255→                <span className="text-muted-foreground">Date</span>
   256→                <span className="font-medium">{selectedDate && format(selectedDate, "EEEE d MMMM yyyy", { locale: fr })}</span>
   257→              </div>
   258→              <div className="flex justify-between text-sm">
   259→                <span className="text-muted-foreground">Créneau</span>
   260→                <span className="font-medium">{selectedSlot?.label}</span>
   261→              </div>
   262→              <div className="flex justify-between text-sm">
   263→                <span className="text-muted-foreground">Acompte</span>
   264→                <span className="font-medium text-primary">5 000 FCFA</span>
   265→              </div>
   266→            </div>
   267→            <Button className="mt-4 w-full max-w-sm" onClick={() => onOpenChange(false)}>Fermer</Button>

   270→          <>
   271→            <DialogHeader className="p-6 pb-2">
   272→              <DialogTitle className="text-xl font-bold">Réserver un créneau</DialogTitle>
   273→              <DialogDescription>{step === 'select' ? 'Choisissez votre date et votre horaire' : 'Complétez vos informations'}</DialogDescription>
   274→            </DialogHeader>
   275→            <div className="flex-1 overflow-hidden">
   276→              <ScrollArea className="h-full max-h-[60vh]">
   277→                <div className="p-6 pt-2">
   278→                  {step === 'select' && (

   374→                          {isSubmitting ? (

   376→                              <span className="w-4 h-4 border-2 border-primary-foreground/30 border-t-primary-foreground rounded-full animate-spin" />
