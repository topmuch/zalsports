    52→function isToday(date: Date): boolean {
    53→  const now = new Date();
    54→  return date.getFullYear() === now.getFullYear() &&
    55→    date.getMonth() === now.getMonth() &&
    56→    date.getDate() === now.getDate();
    57→}
    58→
    59→function generateTimeSlots(selectedDate?: Date, availableHours?: Set<string>): TimeSlot[] {
    60→  const slots: TimeSlot[] = [];
    61→  const now = new Date();
    62→  const currentHour = now.getHours();
    63→  const today = isToday(selectedDate || new Date());
    64→
    65→  for (let h = 8; h <= 23; h++) {
    66→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    67→    const booked = availableHours ? availableHours.has(timeStr) : false;
    68→    const past = today && h <= currentHour;
    69→    const available = !booked && !past;
    70→    slots.push({
    71→      id: `slot-${h}`,
    72→      time: timeStr,
    73→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
    74→      available,
    75→      past,
    76→      price: 25000,
    77→    });
    78→  }
    79→  return slots;
    80→}

    85→
    86→function SlotGridButton({ slot, selected, onClick }: { slot: TimeSlot; selected: boolean; onClick: () => void }) {
    87→  const pastStyle = 'bg-gray-100 border-2 border-gray-200 text-gray-400 cursor-not-allowed';
    88→  const bookedStyle = 'bg-red-50 border-2 border-red-300 text-red-400 cursor-not-allowed line-through';
    89→  const selectedStyle = 'bg-green-800 text-white border-2 border-green-800 shadow-lg shadow-green-800/30 scale-105';
    90→  const availableStyle = 'bg-green-50 border-2 border-green-700 text-green-900 hover:bg-green-100 hover:border-green-800';
    91→
    92→  const style = slot.past ? pastStyle : !slot.available ? bookedStyle : selected ? selectedStyle : availableStyle;
    93→
    94→  return (
    95→    <button
    96→      disabled={!slot.available}
    97→      onClick={onClick}
    98→      className={`relative rounded-xl p-3 text-sm font-bold transition-all duration-200 ${style}`}
    99→    >
   100→      {slot.time}
   101→      {slot.past && (
   102→        <span className="absolute top-1.5 right-1.5 text-[9px] text-gray-400 font-medium">Passé</span>
   103→      )}
   104→      {!slot.available && !slot.past && (
   105→        <span className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-red-400" />
   106→      )}
   107→      {selected && (
   108→        <Check className="absolute top-1.5 right-1.5 w-3.5 h-3.5 text-white" />
   109→      )}
   110→    </button>
   111→  );
   112→}
