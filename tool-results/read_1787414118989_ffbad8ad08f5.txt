    96→function generateTimeSlots(): TimeSlot[] {
    97→  const slots: TimeSlot[] = [];
    98→  for (let h = 8; h <= 23; h++) {
    99→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   100→    const available = Math.random() > 0.4;
   101→    slots.push({
   102→      id: `slot-${h}`,
   103→      time: timeStr,
   104→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   105→      available,
   106→      price: 25000,
   107→    });
   108→  }
   109→  return slots;
   110→}

   112→/* ───────────── Animated Counter ───────────── */
   113→function AnimatedCounter({
   114→  value,
   115→  suffix,
   116→  inView,
   117→}: {
   118→  value: number;
   119→  suffix: string;
   120→  inView: boolean;
   121→}) {
   122→  const [count, setCount] = useState(0);
   123→
   124→  useEffect(() => {
   125→    if (!inView) return;
   126→    let start = 0;
   127→    const duration = 2000;
   128→    const step = Math.max(1, Math.floor(value / (duration / 16)));
   129→    const timer = setInterval(() => {
   130→      start += step;
   131→      if (start >= value) {
   132→        setCount(value);
   133→        clearInterval(timer);
   134→      } else {
   135→        setCount(start);
   136→      }
   137→    }, 16);
   138→    return () => clearInterval(timer);
   139→  }, [inView, value]);
   140→
   141→  return (
   142→    <span>
   143→      {count.toLocaleString('fr-FR')}
   144→      {suffix}
   145→    </span>
   146→  );
   147→}

   149→/* ───────────── Section Wrapper ───────────── */
   150→function Section({
   151→  children,
   152→  className = '',
   153→  id,
   154→}: {
   155→  children: React.ReactNode;
   156→  className?: string;
   157→  id?: string;
   158→}) {
   159→  const ref = useRef<HTMLDivElement>(null);
   160→  const isInView = useInView(ref, { once: true, margin: '-80px' });
   161→
   162→  return (
   163→    <motion.section
   164→      ref={ref}
   165→      id={id}
   166→      className={className}
   167→      initial={{ opacity: 0, y: 40 }}
   168→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   169→      transition={{ duration: 0.7, ease: 'easeOut' }}
   170→    >
   171→      {children}
   172→    </motion.section>
   173→  );
   174→}
