    92→    92→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    93→    93→  const slots: TimeSlot[] = [];
    94→    94→  for (let h = 8; h <= 23; h++) {
    95→    95→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    96→    96→    const available = availableHours ? !availableHours.has(timeStr) : true;
    97→    97→    slots.push({
    98→    98→      id: `slot-${h}`,
    99→    99→      time: timeStr,
   100→   100→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   101→   101→      available,
   102→   102→      price: 25000,
   103→   103→    });
   104→   104→  }
   105→   105→  return slots;
   106→   106→}

   111→   111→
   112→   112→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   113→   113→  const [count, setCount] = useState(0);
   114→   114→  useEffect(() => {
   115→   115→    if (!inView) return;
   116→   116→    let start = 0;
   117→   117→    const step = Math.max(1, Math.floor(value / 125));
   118→   118→    const timer = setInterval(() => {
   119→   119→      start += step;
   120→   120→      if (start >= value) {
   121→   121→        setCount(value);
   122→   122→        clearInterval(timer);
   123→   123→      } else {
   124→   124→        setCount(start);
   125→   125→      }
   126→   126→    }, 16);
   127→   127→    return () => clearInterval(timer);
   128→   128→  }, [inView, value]);
   129→   129→
   130→   130→  return (
   131→   131→    <span>
   132→   132→      {count.toLocaleString('fr-FR')}
   133→   133→      {suffix}
   134→   134→    </span>
   135→   135→  );
   136→   136→}
