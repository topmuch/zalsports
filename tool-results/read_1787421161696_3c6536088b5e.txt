    94→    94→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    95→    95→  const slots: TimeSlot[] = [];
    96→    96→  for (let h = 8; h <= 23; h++) {
    97→    97→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    98→    98→    const available = availableHours ? !availableHours.has(timeStr) : true;
    99→    99→    slots.push({
   100→   100→      id: `slot-${h}`,
   101→   101→      time: timeStr,
   102→   102→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   103→   103→      available,
   104→   104→      price: 25000,
   105→   105→    });
   106→   106→  }
   107→   107→  return slots;
   108→   108→}

   113→   113→
   114→   114→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   115→   115→  const [count, setCount] = useState(0);
   116→   116→  useEffect(() => {
   117→   117→    if (!inView) return;
   118→   118→    let start = 0;
   119→   119→    const step = Math.max(1, Math.floor(value / 125));
   120→   120→    const timer = setInterval(() => {
   121→   121→      start += step;
   122→   122→      if (start >= value) {
   123→   123→        setCount(value);
   124→   124→        clearInterval(timer);
   125→   125→      } else {
   126→   126→        setCount(start);
   127→   127→      }
   128→   128→    }, 16);
   129→   129→    return () => clearInterval(timer);
   130→   130→  }, [inView, value]);
   131→   131→
   132→   132→  return (
   133→   133→    <span>
   134→   134→      {count.toLocaleString('fr-FR')}
   135→   135→      {suffix}
   136→   136→    </span>
   137→   137→  );
   138→   138→}
