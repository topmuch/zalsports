    92→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    93→  const slots: TimeSlot[] = [];
    94→  for (let h = 8; h <= 23; h++) {
    95→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    96→    const available = availableHours ? !availableHours.has(timeStr) : true;
    97→    slots.push({
    98→      id: `slot-${h}`,
    99→      time: timeStr,
   100→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   101→      available,
   102→      price: 25000,
   103→    });
   104→  }
   105→  return slots;
   106→}

   108→/* ═══════════════════════════════════════════
   109→   Animated Counter
   110→   ═══════════════════════════════════════════ */
   111→
   112→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   113→  const [count, setCount] = useState(0);
   114→  useEffect(() => {
   115→    if (!inView) return;
   116→    let start = 0;
   117→    const step = Math.max(1, Math.floor(value / 125));
   118→    const timer = setInterval(() => {
   119→      start += step;
   120→      if (start >= value) {
   121→        setCount(value);
   122→        clearInterval(timer);
   123→      } else {
   124→        setCount(start);
   125→      }
   126→    }, 16);
   127→    return () => clearInterval(timer);
   128→  }, [inView, value]);
   129→
   130→  return (
   131→    <span>
   132→      {count.toLocaleString('fr-FR')}
   133→      {suffix}
   134→    </span>
   135→  );
   136→}

   142→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   143→  const ref = useRef<HTMLDivElement>(null);
   144→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   145→  return (
   146→    <motion.section
   147→      ref={ref}
   148→      id={id}
   149→      className={className}
   150→      initial={{ opacity: 0, y: 30 }}
   151→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   152→      transition={{ duration: 0.6, ease: 'easeOut' }}
   153→    >
   154→      {children}
   155→    </motion.section>
   156→  );
   157→}

   168→  const [name, setName] = useState('');
   169→  const [phone, setPhone] = useState('');
   170→  const [isSubmitting, setIsSubmitting] = useState(false);
   171→  const [loadingSlots, setLoadingSlots] = useState(false);
   172→
   173→  const fetchSlots = useCallback(async (date: Date) => {
   174→    setLoadingSlots(true);
   175→    try {
   176→      const dateStr = format(date, 'yyyy-MM-dd');
   177→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   178→      if (res.ok) {
   179→        const data = await res.json();
   180→        const bookedSet = new Set(
   181→          data.available
   182→            .filter((s: { time: string; available: boolean }) => !s.available)
   183→            .map((s: { time: string; available: boolean }) => s.time)
   184→        );
   185→        setSlots(generateTimeSlots(bookedSet));
   186→      } else {
   187→        setSlots(generateTimeSlots());
   188→      }
   189→    } catch {
   190→      setSlots(generateTimeSlots());
   191→    } finally {
   192→      setLoadingSlots(false);
   193→    }
   194→  }, []);
   195→
   196→  useEffect(() => {
   197→    if (open) {
   198→      const tomorrow = addDays(new Date(), 1);
   199→      setSelectedDate(tomorrow);
   200→      setSelectedSlot(null);
   201→      setStep('select');
   202→      setName('');
   203→      setPhone('');
   204→      fetchSlots(tomorrow);
   205→    }
   206→  }, [open, fetchSlots]);
   207→

   209→    if (open && selectedDate) {

   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify({
   223→          date: format(selectedDate, 'yyyy-MM-dd'),
   224→          timeSlot: selectedSlot.time,
   225→          customerName: name,
   226→          customerPhone: phone,
   227→        }),
   228→      });
   229→      if (res.ok) setStep('confirm');
   230→    } catch {
   231→      // silent
   232→    } finally {
   233→      setIsSubmitting(false);
   234→    }
   235→  }, [selectedSlot, selectedDate, name, phone]);
   236→
   237→  return (
   238→    <Dialog open={open} onOpenChange={onOpenChange}>
   239→      <DialogContent className="sm:max-w-2xl bg-card border-border max-h-[90vh] overflow-hidden flex flex-col p-0">
   240→        {step === 'confirm' ? (

   242→            <div className="w-16 h-16 rounded-full bg-primary/20 flex items-center justify-center">
   243→              <Zap className="w-8 h-8 text-primary" />
   244→            </div>
   245→            <DialogHeader>
   246→              <DialogTitle className="text-2xl">Réservation confirmée !</DialogTitle>
   247→              <DialogDescription className="text-muted-foreground text-base mt-2">
   248→                Votre créneau a été réservé avec succès. Vous recevrez un SMS de confirmation.
   249→              </DialogDescription>
   250→            </DialogHeader>
   251→            <div className="bg-secondary rounded-xl p-4 mt-4 w-full max-w-sm space-y-2">
   252→              <div className="flex justify-between text-sm">
   253→                <span className="text-muted-foreground">Date</span>
   254→                <span className="font-medium">{selectedDate && format(selectedDate, "EEEE d MMMM yyyy", { locale: fr })}</span>
   255→              </div>
   256→              <div className="flex justify-between text-sm">
   257→                <span className="text-muted-foreground">Créneau</span>
   258→                <span className="font-medium">{selectedSlot?.label}</span>
   259→              </div>
   260→              <div className="flex justify-between text-sm">
   261→                <span className="text-muted-foreground">Acompte</span>

   271→              <DialogDescription>{step === 'select' ? 'Choisissez votre date et votre horaire' : 'Complétez vos informations'}</DialogDescription>
   272→            </DialogHeader>
   273→            <div className="flex-1 overflow-hidden">
   274→              <ScrollArea className="h-full max-h-[60vh]">
