   172→
   173→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   174→  const slots: TimeSlot[] = [];
   175→  for (let h = 8; h <= 23; h++) {
   176→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   177→    const available = availableHours ? !availableHours.has(timeStr) : true;
   178→    slots.push({
   179→      id: `slot-${h}`,
   180→      time: timeStr,
   181→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   182→      available,
   183→      price: 25000,
   184→    });
   185→  }
   186→  return slots;
   187→}

   189→/* ═══════════════════════════════════════════
   190→   Animated Counter
   191→   ═══════════════════════════════════════════ */
   192→
   193→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   194→  const [count, setCount] = useState(0);
   195→  useEffect(() => {
   196→    if (!inView) return;
   197→    let start = 0;
   198→    const step = Math.max(1, Math.floor(value / 125));
   199→    const timer = setInterval(() => {
   200→      start += step;
   201→      if (start >= value) {
   202→        setCount(value);
   203→        clearInterval(timer);
   204→      } else {
   205→        setCount(start);
   206→      }
   207→    }, 16);
   208→    return () => clearInterval(timer);
   209→  }, [inView, value]);
   210→
   211→  return (
   212→    <span>
   213→      {count.toLocaleString('fr-FR')}
   214→      {suffix}
   215→    </span>
   216→  );
   217→}

   222→
   223→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   224→  const ref = useRef<HTMLDivElement>(null);
   225→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   226→  return (
   227→    <motion.section
   228→      ref={ref}
   229→      id={id}
   230→      className={className}
   231→      initial={{ opacity: 0, y: 30 }}
   232→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   233→      transition={{ duration: 0.6, ease: 'easeOut' }}
   234→    >
   235→      {children}
   236→    </motion.section>
   237→  );
   238→}

   248→  const [step, setStep] = useState<'select' | 'info' | 'payment' | 'confirm'>('select');
   249→  const [name, setName] = useState('');
   250→  const [phone, setPhone] = useState('');
   251→  const [isSubmitting, setIsSubmitting] = useState(false);
   252→  const [loadingSlots, setLoadingSlots] = useState(false);
   253→  const [paymentMethod, setPaymentMethod] = useState<'wave' | 'orange_money' | null>(null);
   254→  const [paymentPhone, setPaymentPhone] = useState('');
   255→  const [bookingId, setBookingId] = useState<string | null>(null);
   256→  const [isPaying, setIsPaying] = useState(false);
   257→  const [confirmingPayment, setConfirmingPayment] = useState(false);
   258→
   259→  const fetchSlots = useCallback(async (date: Date) => {
   260→    setLoadingSlots(true);
   261→    try {
   262→      const dateStr = format(date, 'yyyy-MM-dd');
   263→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   264→      if (res.ok) {
   265→        const data = await res.json();
   266→        const bookedSet = new Set(
   267→          data.available
   268→            .filter((s: { time: string; available: boolean }) => !s.available)
   269→            .map((s: { time: string; available: boolean }) => s.time)
   270→        );
   271→        setSlots(generateTimeSlots(bookedSet));
   272→      } else {
   273→        setSlots(generateTimeSlots());
   274→      }
   275→    } catch {
   276→      setSlots(generateTimeSlots());
   277→    } finally {
   278→      setLoadingSlots(false);
   279→    }
   280→  }, []);
   281→
   282→  useEffect(() => {
   283→    if (open) {
   284→      const tomorrow = addDays(new Date(), 1);
   285→      setSelectedDate(tomorrow);
   286→      setSelectedSlot(null);
   287→      setStep('select');
   288→      setName('');
   289→      setPhone('');
   290→      setPaymentMethod(null);
   291→      setPaymentPhone('');
   292→      setBookingId(null);
   293→      setIsPaying(false);
   294→      setConfirmingPayment(false);
   295→      fetchSlots(tomorrow);
   296→    }
   297→  }, [open, fetchSlots]);
   298→
   299→  useEffect(() => {
   300→    if (open && selectedDate) {
   301→      setSelectedSlot(null);
   302→      fetchSlots(selectedDate);
   303→    }
   304→  }, [selectedDate, open, fetchSlots]);
   305→
   306→  const handleConfirmInfo = useCallback(async () => {
   307→    if (!selectedSlot || !selectedDate || !name || !phone) return;
   308→    setIsSubmitting(true);
   309→    try {
   310→      const res = await fetch('/api/bookings', {
   311→        method: 'POST',
   312→        headers: { 'Content-Type': 'application/json' },
   313→        body: JSON.stringify({
   314→          date: format(selectedDate, 'yyyy-MM-dd'),
   315→          timeSlot: selectedSlot.time,
   316→          customerName: name,
   317→          customerPhone: phone,
   318→        }),
   319→      });
   320→      if (res.ok) {
   321→        const data = await res.json();
   322→        setBookingId(data.id || data.booking?.id || null);
   323→        setPaymentPhone(phone);
   324→        setStep('payment');
   325→      }
   326→    } catch {
   327→      // silent
   328→    } finally {
   329→      setIsSubmitting(false);
   330→    }
   331→  }, [selectedSlot, selectedDate, name, phone]);
   332→
   333→  const handleInitiatePayment = useCallback(async () => {
   334→    if (!paymentMethod || !paymentPhone || !bookingId) return;
   335→    setIsPaying(true);
   336→    try {
   337→      const res = await fetch('/api/payments/initiate', {
   338→        method: 'POST',
   339→        headers: { 'Content-Type': 'application/json' },
   340→        body: JSON.stringify({
   341→          bookingId,
   342→          method: paymentMethod,
   343→          phone: paymentPhone,
   344→          amount: 5000,
   345→        }),
   346→      });
   347→      if (res.ok) {
   348→        const data = await res.json();
   349→        if (data.paymentUrl) {
   350→          window.open(data.paymentUrl, '_blank');
   351→        }
   352→        setStep('confirm');
   353→      }
   354→    } catch {
   355→      // silent
   356→    } finally {
   357→      setIsPaying(false);
   358→    }
   359→  }, [paymentMethod, paymentPhone, bookingId]);
   360→
   361→  const handleConfirmPaid = useCallback(async () => {
   362→    if (!bookingId || !paymentMethod) return;
   363→    setConfirmingPayment(true);

   369→          paymentStatus: 'partial',
   370→          paymentMethod,
   371→          depositPaid: 5000,
   372→        }),
   373→      });
   374→      setStep('confirm');
   375→    } catch {
   376→      // silent
   377→    } finally {
   378→      setConfirmingPayment(false);
   379→    }
   380→  }, [bookingId, paymentMethod]);
   381→
   382→  return (
   383→    <Dialog open={open} onOpenChange={onOpenChange}>
   384→      <DialogContent className="sm:max-w-2xl md:max-w-3xl bg-card border-border max-h-[95vh] overflow-hidden flex flex-col p-0">
   385→        {step === 'confirm' ? (

   393→                Votre créneau a été réservé avec succès. Vous recevrez un SMS de confirmation.
   394→              </DialogDescription>
   395→            </DialogHeader>
   396→            <div className="bg-secondary rounded-xl p-4 mt-4 w-full max-w-sm space-y-2">
   397→              <div className="flex justify-between text-sm">
   398→                <span className="text-muted-foreground">Date</span>
   399→                <span className="font-medium">{selectedDate && format(selectedDate, "EEEE d MMMM yyyy", { locale: fr })}</span>
   400→              </div>
   401→              <div className="flex justify-between text-sm">
   402→                <span className="text-muted-foreground">Créneau</span>
   403→                <span className="font-medium">{selectedSlot?.label}</span>
   404→              </div>
   405→              <div className="flex justify-between text-sm">
   406→                <span className="text-muted-foreground">Méthode de paiement</span>
