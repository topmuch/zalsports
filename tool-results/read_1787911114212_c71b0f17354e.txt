   176→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   177→  const slots: TimeSlot[] = [];
   178→  for (let h = 8; h <= 23; h++) {
   179→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   180→    const available = availableHours ? !availableHours.has(timeStr) : true;
   181→    slots.push({
   182→      id: `slot-${h}`,
   183→      time: timeStr,
   184→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   185→      available,
   186→      price: 25000,
   187→    });
   188→  }
   189→  return slots;
   190→}

   192→/* ═══════════════════════════════════════════
   193→   Animated Counter
   194→   ═══════════════════════════════════════════ */
   195→
   196→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   197→  const [count, setCount] = useState(0);
   198→  useEffect(() => {
   199→    if (!inView) return;
   200→    let start = 0;
   201→    const step = Math.max(1, Math.floor(value / 125));
   202→    const timer = setInterval(() => {
   203→      start += step;
   204→      if (start >= value) {
   205→        setCount(value);
   206→        clearInterval(timer);
   207→      } else {
   208→        setCount(start);
   209→      }
   210→    }, 16);
   211→    return () => clearInterval(timer);
   212→  }, [inView, value]);
   213→
   214→  return (
   215→    <span>
   216→      {count.toLocaleString('fr-FR')}
   217→      {suffix}
   218→    </span>
   219→  );
   220→}

   225→
   226→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   227→  const ref = useRef<HTMLDivElement>(null);
   228→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   229→  return (
   230→    <motion.section
   231→      ref={ref}
   232→      id={id}
   233→      className={className}
   234→      initial={{ opacity: 0, y: 30 }}
   235→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   236→      transition={{ duration: 0.6, ease: 'easeOut' }}
   237→    >
   238→      {children}
   239→    </motion.section>
   240→  );
   241→}

   247→function BookingDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   248→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(addDays(new Date(), 1));
   249→  const [selectedSlot, setSelectedSlot] = useState<TimeSlot | null>(null);
   250→  const [slots, setSlots] = useState<TimeSlot[]>([]);
   251→  const [step, setStep] = useState<'select' | 'info' | 'payment' | 'confirm'>('select');
   252→  const [name, setName] = useState('');
   253→  const [phone, setPhone] = useState('');
   254→  const [isSubmitting, setIsSubmitting] = useState(false);
   255→  const [loadingSlots, setLoadingSlots] = useState(false);
   256→  const [paymentMethod, setPaymentMethod] = useState<'wave' | 'orange_money' | 'cash' | null>(null);
   257→  const [paymentPhone, setPaymentPhone] = useState('');
   258→  const [bookingId, setBookingId] = useState<string | null>(null);
   259→  const [isPaying, setIsPaying] = useState(false);
   260→
   261→  const fetchSlots = useCallback(async (date: Date) => {
   262→    setLoadingSlots(true);
   263→    try {
   264→      const dateStr = format(date, 'yyyy-MM-dd');
   265→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   266→      if (res.ok) {
   267→        const data = await res.json();
   268→        const bookedSet = new Set(
   269→          data.available
   270→            .filter((s: { time: string; available: boolean }) => !s.available)
   271→            .map((s: { time: string; available: boolean }) => s.time)
   272→        );
   273→        setSlots(generateTimeSlots(bookedSet));
   274→      } else {
   275→        setSlots(generateTimeSlots());
   276→      }
   277→    } catch {
   278→      setSlots(generateTimeSlots());
   279→    } finally {
   280→      setLoadingSlots(false);
   281→    }
   282→  }, []);
   283→
   284→  useEffect(() => {
   285→    if (open) {
   286→      const tomorrow = addDays(new Date(), 1);
   287→      setSelectedDate(tomorrow);
   288→      setSelectedSlot(null);
   289→      setStep('select');
   290→      setName('');
   291→      setPhone('');
   292→      setPaymentMethod(null);
   293→      setPaymentPhone('');
   294→      setBookingId(null);
   295→      setIsPaying(false);
   296→      fetchSlots(tomorrow);
   297→    }
   298→  }, [open, fetchSlots]);
   299→
   300→  useEffect(() => {
   301→    if (open && selectedDate) {
   302→      setSelectedSlot(null);
   303→      fetchSlots(selectedDate);
   304→    }
   305→  }, [selectedDate, open, fetchSlots]);
   306→
   307→  const handleConfirmInfo = useCallback(async () => {
   308→    if (!selectedSlot || !selectedDate || !name || !phone) return;
   309→    setIsSubmitting(true);
   310→    try {
   311→      const res = await fetch('/api/bookings', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({
   315→          date: format(selectedDate, 'yyyy-MM-dd'),
   316→          timeSlot: selectedSlot.time,
   317→          customerName: name,
   318→          customerPhone: phone,
   319→        }),
   320→      });
   321→      if (res.ok) {
   322→        const data = await res.json();
   323→        setBookingId(data.id || data.booking?.id || null);
   324→        setPaymentPhone(phone);
   325→        setStep('payment');
   326→      }
   327→    } catch {
   328→      // silent
   329→    } finally {
   330→      setIsSubmitting(false);
   331→    }
   332→  }, [selectedSlot, selectedDate, name, phone]);
   333→
   334→  const handleInitiatePayment = useCallback(async () => {
   335→    if (!paymentMethod || !bookingId) return;
   336→    if (paymentMethod !== 'cash' && !paymentPhone) return;
   337→
   338→    setIsPaying(true);
   339→    try {
   340→      // For all methods: directly confirm the booking with payment info
   341→      // In production, Wave/OM would redirect to a real payment gateway
   342→      await fetch(`/api/bookings/${bookingId}`, {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({
   346→          status: 'confirmed',
   347→          paymentStatus: paymentMethod === 'cash' ? 'pending' : 'partial',
   348→          paymentMethod,
   349→          depositPaid: paymentMethod === 'cash' ? 0 : 5000,
   350→        }),
   351→      });
   352→      setStep('confirm');
   353→    } catch {
   354→      // silent
   355→    } finally {
   356→      setIsPaying(false);
   357→    }
   358→  }, [paymentMethod, paymentPhone, bookingId]);
   359→
   360→  return (
   361→    <Dialog open={open} onOpenChange={onOpenChange}>
   362→      <DialogContent className="sm:max-w-2xl md:max-w-3xl bg-card border-border max-h-[95vh] overflow-hidden flex flex-col p-0">
   363→        {step === 'confirm' ? (
   364→          <div className="flex-1 flex flex-col items-center justify-center p-8 gap-4 text-center">
   365→            <div className="w-16 h-16 rounded-full bg-primary/20 flex items-center justify-center">
   366→              <Zap className="w-8 h-8 text-primary" />
   367→            </div>
   368→            <DialogHeader>
   369→              <DialogTitle className="text-2xl">Réservation confirmée !</DialogTitle>
   370→              <DialogDescription className="text-muted-foreground text-base mt-2">
   371→                Votre créneau a été réservé avec succès. Vous recevrez un SMS de confirmation.
   372→              </DialogDescription>
   373→            </DialogHeader>
   374→            <div className="bg-secondary rounded-xl p-4 mt-4 w-full max-w-sm space-y-2">
   375→              <div className="flex justify-between text-sm">
   376→                <span className="text-muted-foreground">Date</span>
   377→                <span className="font-medium">{selectedDate && format(selectedDate, "EEEE d MMMM yyyy", { locale: fr })}</span>
   378→              </div>
   379→              <div className="flex justify-between text-sm">
   380→                <span className="text-muted-foreground">Créneau</span>
   381→                <span className="font-medium">{selectedSlot?.label}</span>
   382→              </div>
   383→              <div className="flex justify-between text-sm">
   384→                <span className="text-muted-foreground">Méthode de paiement</span>

   390→              </div>
   391→            </div>
   392→            <div className="bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-800 rounded-xl p-3 w-full max-w-sm mt-1">
   393→              <p className="text-sm text-emerald-700 dark:text-emerald-400 font-medium">Réservation confirmée</p>
