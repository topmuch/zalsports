   106→function statusBadge(status: Booking['status']) {
   107→  switch (status) {
   108→    case 'pending':

   110→    case 'confirmed':
   111→      return <Badge className="bg-primary/15 text-primary border-primary/30">Confirmé</Badge>;
   112→    case 'completed':
   113→      return <Badge className="bg-emerald-500/15 text-emerald-600 border-emerald-500/30">Terminé</Badge>;

   142→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   143→  const slots: TimeSlot[] = [];
   144→  for (let h = 8; h <= 23; h++) {
   145→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   146→    slots.push({
   147→      id: `slot-${h}`,
   148→      time: timeStr,
   149→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   150→      available: availableHours ? !availableHours.has(timeStr) : true,
   151→    });
   152→  }
   153→  return slots;
   154→}

   160→function PhoneLogin({ onLogin }: { onLogin: (phone: string) => void }) {
   161→  const [phone, setPhone] = useState('');
   162→  const [loading, setLoading] = useState(false);
   163→  const [error, setError] = useState('');
   164→
   165→  const handleSubmit = async (e: React.FormEvent) => {
   166→    e.preventDefault();
   167→    setError('');
   168→    if (!/^\+?221\s?\d{2}\s?\d{3}\s?\d{2}\s?\d{2}$/.test(phone.trim())) {
   169→      setError('Format invalide. Ex: 78 278 49 49');
   170→      return;
   171→    }
   172→    setLoading(true);
   173→    // Try to find profile, if not found create one on the fly
   174→    try {
   175→      const res = await fetch(`/api/user/profile?phone=${encodeURIComponent(phone.trim())}`);
   176→      if (res.ok || res.status === 404) {
   177→        onLogin(phone.trim());
   178→      } else {
   179→        setError('Erreur serveur. Réessayez.');
   180→      }
   181→    } catch {
   182→      setError('Erreur de connexion.');
   183→    } finally {
   184→      setLoading(false);
   185→    }
   186→  };
   187→
   188→  return (
   189→    <div className="min-h-[70vh] flex items-center justify-center px-4">
   190→      <motion.div
   191→        initial={{ opacity: 0, y: 20 }}
   192→        animate={{ opacity: 1, y: 0 }}
   193→        className="w-full max-w-sm space-y-6 text-center"
   194→      >
   195→        <div className="w-16 h-16 rounded-2xl bg-primary/10 flex items-center justify-center mx-auto">
   196→          <Phone className="w-8 h-8 text-primary" />
   197→        </div>
   198→        <div>
   199→          <h2 className="text-2xl font-bold">Mon espace</h2>
   200→          <p className="text-sm text-muted-foreground mt-2">
   201→            Entrez votre numéro pour accéder à vos réservations et paramètres.
   202→          </p>
   203→        </div>
   204→        <form onSubmit={handleSubmit} className="space-y-4">
   205→          <div>
   206→            <Label htmlFor="login-phone" className="sr-only">Téléphone</Label>
   207→            <Input
   208→              id="login-phone"
   209→              placeholder="78 278 49 49"
   210→              value={phone}
   211→              onChange={(e) => setPhone(e.target.value)}
   212→              className="text-center text-lg h-12"
   213→              autoFocus
   214→            />
   215→          </div>
   216→          {error && <p className="text-xs text-destructive">{error}</p>}
   217→          <Button type="submit" className="w-full h-12" disabled={loading || !phone}>
   218→            {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : <><Phone className="w-4 h-4 mr-2" /> Accéder</>}
   219→          </Button>
   220→        </form>
   221→        <p className="text-xs text-muted-foreground">
   222→          Aucun compte requis — votre numéro vous identifie.
   223→        </p>
   224→      </motion.div>
   225→    </div>
   226→  );
   227→}

   233→function CalendarTab({ phone }: { phone: string }) {
   234→  const [currentMonth, setCurrentMonth] = useState(new Date());
   235→  const [calendarData, setCalendarData] = useState<CalendarDay[]>([]);
   236→  const [selectedDate, setSelectedDate] = useState<Date | undefined>();
   237→  const [loading, setLoading] = useState(false);
   238→
   239→  const fetchCalendar = useCallback(async (date: Date) => {
   240→    setLoading(true);
   241→    try {
   242→      const y = date.getFullYear();
   243→      const m = date.getMonth() + 1;
   244→      const res = await fetch(`/api/user/calendar?phone=${encodeURIComponent(phone)}&year=${y}&month=${m}`);
   245→      if (res.ok) {
   246→        const data = await res.json();
   247→        setCalendarData(data.days || []);
   248→      }
   249→    } catch { /* silent */ } finally {
   250→      setLoading(false);
   251→    }
   252→  }, [phone]);
   253→
   254→  useEffect(() => { fetchCalendar(currentMonth); }, [currentMonth, fetchCalendar]);
   255→
   256→  const bookingDates = new Set(calendarData.map(d => d.date));
   257→  const selectedDayData = selectedDate ? calendarData.find(d => d.date === format(selectedDate, 'yyyy-MM-dd')) : null;
   258→
   259→  return (
   260→    <div className="space-y-6">
   261→      {/* Month Navigation */}
   262→      <div className="flex items-center justify-between">
   263→        <Button variant="ghost" size="sm" onClick={() => setCurrentMonth(new Date(currentMonth.getFullYear(), currentMonth.getMonth() - 1, 1))}>
   264→          <ChevronLeft className="w-4 h-4" />
   265→        </Button>
   266→        <h3 className="text-sm font-semibold">
   267→          {format(currentMonth, 'MMMM yyyy', { locale: fr })}
   268→        </h3>
   269→        <Button variant="ghost" size="sm" onClick={() => setCurrentMonth(new Date(currentMonth.getFullYear(), currentMonth.getMonth() + 1, 1))}>
   270→          <ChevronRight className="w-4 h-4" />
   271→        </Button>
   272→      </div>
   273→
   274→      {/* Calendar Grid */}
   275→      <div className="flex justify-center">
   276→        <Calendar
   277→          mode="single"
   278→          selected={selectedDate}
   279→          onSelect={setSelectedDate}
   280→          month={currentMonth}
   281→          onMonthChange={setCurrentMonth}
   282→          modifiers={{
   283→            hasBookings: (date) => bookingDates.has(format(date, 'yyyy-MM-dd')),
   284→          }}
   285→          modifiersClassNames={{
   286→            hasBookings: 'bg-primary/20 font-bold',
   287→          }}
   288→          className="rounded-xl border border-border"
   289→          classNames={{
   290→            day_selected: 'bg-primary text-primary-foreground rounded-md',
   291→            day_today: 'bg-accent text-accent-foreground rounded-md',
   292→          }}
   293→        />
   294→      </div>
   295→
   296→      {/* Selected Day Detail */}
   297→      {selectedDate && (
   298→        <AnimatePresence mode="wait">
   299→          <motion.div
   300→            key={format(selectedDate, 'yyyy-MM-dd')}
   301→            initial={{ opacity: 0, y: 10 }}
   302→            animate={{ opacity: 1, y: 0 }}
   303→            exit={{ opacity: 0, y: -10 }}
   304→            className="bg-secondary/50 rounded-xl p-4 space-y-3"
   305→          >
   306→            <h4 className="text-sm font-semibold">
   307→              {format(selectedDate, 'EEEE d MMMM yyyy', { locale: fr })}
   308→            </h4>
   309→            {selectedDayData && selectedDayData.bookings.length > 0 ? (
   310→              <div className="space-y-2">
   311→                {selectedDayData.bookings.map((b) => (
   312→                  <div key={b.id} className="flex items-center justify-between bg-card rounded-lg p-3 border border-border">
   313→                    <div className="flex items-center gap-3">
   314→                      <Clock className="w-4 h-4 text-primary" />
   315→                      <span className="text-sm font-mono font-medium">{b.timeSlot}</span>
   316→                    </div>
   317→                    <div className="flex items-center gap-2">
   318→                      {statusBadge(b.status as Booking['status'])}
   319→                      {paymentBadge(b.paymentStatus as Booking['paymentStatus'])}
   320→                    </div>
   321→                  </div>
   322→                ))}
   323→              </div>
   324→            ) : (
   325→              <p className="text-sm text-muted-foreground text-center py-4">
   326→                Aucune réservation ce jour.
   327→              </p>

   329→          </motion.div>

   335→        <span className="flex items-center gap-1.5"><span className="w-2.5 h-2.5 rounded-full bg-primary/20 border border-primary/40" /> Jour avec réservation</span>
   336→      </div>

   345→function ReservationsTab({ phone }: { phone: string }) {
   346→  const [bookings, setBookings] = useState<Booking[]>([]);
   347→  const [filter, setFilter] = useState<ReservationFilter>('active');
   348→  const [loading, setLoading] = useState(true);
   349→
   350→  const fetchBookings = useCallback(async () => {
   351→    setLoading(true);
   352→    try {
   353→      const params = new URLSearchParams({ phone, limit: '50' });
   354→      const res = await fetch(`/api/user/bookings?${params}`);
   355→      if (res.ok) {
   356→        const data = await res.json();
   357→        setBookings(data.bookings);
   358→      }
   359→    } catch { /* silent */ } finally {
   360→      setLoading(false);
   361→    }
   362→  }, [phone]);
   363→
   364→  useEffect(() => { fetchBookings(); }, [fetchBookings]);
   365→
   366→  const handleCancel = async (id: string) => {
   367→    try {
   368→      const res = await fetch(`/api/user/bookings/${id}?phone=${encodeURIComponent(phone)}`, { method: 'DELETE' });
   369→      if (res.ok) fetchBookings();
   370→    } catch { /* silent */ }
   371→  };
   372→
   373→  const filtered = bookings.filter((b) => {
   374→    if (filter === 'active') return b.status === 'confirmed' || b.status === 'pending';
   375→    if (filter === 'paid') return b.paymentStatus === 'paid' && b.status !== 'cancelled';
   376→    if (filter === 'unpaid') return b.paymentStatus === 'unpaid' && b.status !== 'cancelled';
   377→    return true;
   378→  });
   379→
   380→  const filters: { key: ReservationFilter; label: string; count: number }[] = [
   381→    { key: 'active', label: 'En cours', count: bookings.filter(b => b.status === 'confirmed' || b.status === 'pending').length },
   382→    { key: 'paid', label: 'Payées', count: bookings.filter(b => b.paymentStatus === 'paid' && b.status !== 'cancelled').length },
   383→    { key: 'unpaid', label: 'À payer', count: bookings.filter(b => b.paymentStatus === 'unpaid' && b.status !== 'cancelled').length },
   384→    { key: 'all', label: 'Toutes', count: bookings.length },
   385→  ];
   386→
   387→  return (
   388→    <div className="space-y-5">
   389→      {/* Filter Tabs */}
   390→      <div className="flex bg-secondary rounded-lg p-0.5">
   391→        {filters.map((f) => (
   392→          <button
   393→            key={f.key}
   394→            onClick={() => setFilter(f.key)}
   395→            className={`flex-1 px-3 py-2 text-xs font-medium rounded-md transition-all ${
   396→              filter === f.key
   397→                ? 'bg-primary text-primary-foreground shadow-sm'
   398→                : 'text-muted-foreground hover:text-foreground'
   399→            }`}
   400→          >
   401→            {f.label} <span className="ml-1 opacity-60">({f.count})</span>
   402→          </button>
   403→        ))}
   404→      </div>
   405→
   406→      {/* Booking List */}
   407→      <div className="space-y-3 max-h-[500px] overflow-y-auto custom-scrollbar pr-1">
   408→        {loading ? (
   409→          <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <div key={i} className="h-24 rounded-xl bg-card border border-border animate-pulse" />)}</div>
   410→        ) : filtered.length === 0 ? (
   411→          <div className="text-center py-12 text-muted-foreground">
   412→            <ClipboardList className="w-10 h-10 mx-auto mb-3 opacity-30" />
   413→            <p className="text-sm">Aucune réservation trouvée.</p>
   414→          </div>
   415→        ) : (
   416→          filtered.map((b) => {
   417→            const isPast = isBefore(parseISO(b.date), new Date());
   418→            return (
   419→              <motion.div
   420→                key={b.id}
   421→                initial={{ opacity: 0, y: 10 }}
   422→                animate={{ opacity: 1, y: 0 }}
   423→                className="bg-card border border-border rounded-xl p-4 hover:border-primary/20 transition-colors"
   424→              >
   425→                <div className="flex items-start justify-between gap-3">
   426→                  <div className="space-y-2 flex-1 min-w-0">
   427→                    <div className="flex items-center gap-2 flex-wrap">
   428→                      <span className="text-sm font-mono font-semibold text-primary">{b.timeSlot}</span>
   429→                      {statusBadge(b.status)}
   430→                      {paymentBadge(b.paymentStatus)}
   431→                    </div>
   432→                    <p className="text-sm text-muted-foreground">
   433→                      {format(parseISO(b.date), 'EEEE d MMMM yyyy', { locale: fr })}
   434→                    </p>
   435→                    <div className="flex items-center gap-4 text-xs text-muted-foreground">
   436→                      <span className="flex items-center gap-1"><CreditCard className="w-3 h-3" /> {formatMoney(b.amount)}</span>
   437→                      {b.depositPaid > 0 && <span className="text-emerald-600">Acompte: {formatMoney(b.depositPaid)}</span>}
   438→                    </div>
   439→                  </div>
   440→                  {!isPast && (b.status === 'confirmed' || b.status === 'pending') && (
   441→                    <Button
   442→                      variant="ghost" size="sm" className="h-8 px-2 text-destructive hover:text-destructive hover:bg-destructive/10 shrink-0"
   443→                      onClick={() => handleCancel(b.id)} title="Annuler"
   444→                    >
   445→                      <Trash2 className="w-3.5 h-3.5" />
   446→                    </Button>
   447→                  )}
   448→                </div>
   449→              </motion.div>
   450→            );
   451→          })
   452→        )}
   453→      </div>
   454→    </div>
   455→  );
   456→}

   461→
   462→function NewReservationTab({ phone, name, onCreated }: { phone: string; name: string; onCreated: () => void }) {
   463→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(addDays(new Date(), 1));
   464→  const [selectedSlot, setSelectedSlot] = useState<TimeSlot | null>(null);
   465→  const [slots, setSlots] = useState<TimeSlot[]>([]);
   466→  const [step, setStep] = useState<'select' | 'confirm'>('select');
   467→  const [isSubmitting, setIsSubmitting] = useState(false);
   468→  const [loadingSlots, setLoadingSlots] = useState(false);
   469→  const [success, setSuccess] = useState(false);
   470→
   471→  const fetchSlots = useCallback(async (date: Date) => {
   472→    setLoadingSlots(true);
   473→    try {
   474→      const dateStr = format(date, 'yyyy-MM-dd');
   475→      const res = await fetch(`/api/bookings?date=${dateStr}`);
   476→      if (res.ok) {
   477→        const data = await res.json();
   478→        const bookedSet = new Set(data.available.filter((s: { available: boolean; time: string }) => !s.available).map((s: { time: string }) => s.time));
   479→        setSlots(generateTimeSlots(bookedSet));
   480→      } else {
   481→        setSlots(generateTimeSlots());
   482→      }
   483→    } catch { setSlots(generateTimeSlots()); } finally {
   484→      setLoadingSlots(false);
   485→    }
   486→  }, []);
   487→
   488→  useEffect(() => { fetchSlots(addDays(new Date(), 1)); }, [fetchSlots]);
   489→  useEffect(() => {
   490→    if (selectedDate) { setSelectedSlot(null); fetchSlots(selectedDate); }
   491→  }, [selectedDate, fetchSlots]);
   492→
   493→  const handleSubmit = async () => {
   494→    if (!selectedSlot || !selectedDate) return;
   495→    setIsSubmitting(true);
   496→    try {
   497→      const res = await fetch('/api/bookings', {
   498→        method: 'POST',
   499→        headers: { 'Content-Type': 'application/json' },
   500→        body: JSON.stringify({
   501→          date: format(selectedDate, 'yyyy-MM-dd'),
   502→          timeSlot: selectedSlot.time,
   503→          customerName: name,
   504→          customerPhone: phone,
   505→        }),
   506→      });
   507→      if (res.ok) {
   508→        setSuccess(true);
   509→        onCreated();
   510→      }
   511→    } catch { /* silent */ } finally {
   512→      setIsSubmitting(false);
   513→    }
   514→  };
   515→
   516→  if (success) {
   517→    return (
   518→      <div className="flex flex-col items-center justify-center py-12 gap-4 text-center">
   519→        <div className="w-16 h-16 rounded-full bg-primary/20 flex items-center justify-center">
   520→          <CheckCircle2 className="w-8 h-8 text-primary" />
   521→        </div>
   522→        <div>
   523→          <h3 className="text-lg font-bold">Demande envoyée !</h3>
   524→          <p className="text-sm text-muted-foreground mt-2 max-w-xs">
   525→            Votre réservation a été enregistrée. Vous recevrez une confirmation par SMS.
   526→          </p>
   527→        </div>
   528→        <Button variant="outline" onClick={() => { setSuccess(false); setStep('select'); setSelectedSlot(null); }}>
   529→          Nouvelle réservation
   530→        </Button>
   531→      </div>
   532→    );
   533→  }
   534→

   631→function SettingsTab({ phone, initialName }: { phone: string; initialName: string }) {
   632→  const [name, setName] = useState(initialName);
   633→  const [email, setEmail] = useState('');
   634→  const [notifications, setNotifications] = useState(true);
   635→  const [saving, setSaving] = useState(false);
   636→  const [saved, setSaved] = useState(false);
   637→  const [loading, setLoading] = useState(true);
   638→
   639→  useEffect(() => {
   640→    (async () => {
   641→      try {
   642→        const res = await fetch(`/api/user/profile?phone=${encodeURIComponent(phone)}`);
   643→        if (res.ok) {
   644→          const data = await res.json();
   645→          setName(data.profile.name);
   646→          setEmail(data.profile.email);
   647→          setNotifications(data.profile.notifications);
   648→        }
   649→      } catch { /* silent */ } finally {
   650→        setLoading(false);
   651→      }
   652→    })();
   653→  }, [phone]);
   654→
   655→  const handleSave = async () => {
   656→    setSaving(true);
   657→    setSaved(false);
   658→    try {
   659→      const res = await fetch('/api/user/profile', {
   660→        method: 'PUT',
   661→        headers: { 'Content-Type': 'application/json' },
   662→        body: JSON.stringify({ phone, name, email, notifications }),
   663→      });
   664→      if (res.ok) setSaved(true);
   665→    } catch { /* silent */ } finally {
   666→      setSaving(false);
   667→      setTimeout(() => setSaved(false), 3000);
   668→    }
   669→  };
   670→
   671→  if (loading) {
   672→    return <div className="space-y-4">{Array.from({ length: 3 }).map((_, i) => <div key={i} className="h-16 rounded-xl bg-card border border-border animate-pulse" />)}</div>;
   673→  }
   674→
   675→  return (
   676→    <div className="space-y-6">
   677→      <div>
   678→        <h3 className="text-sm font-semibold mb-4">Informations personnelles</h3>
   679→        <div className="space-y-4">
   680→          <div>
   681→            <Label htmlFor="settings-name">Nom complet</Label>
   682→            <Input id="settings-name" value={name} onChange={(e) => setName(e.target.value)} className="mt-1.5" placeholder="Votre nom" />
   683→          </div>
   684→          <div>
   685→            <Label htmlFor="settings-phone">Téléphone</Label>
   686→            <Input id="settings-phone" value={phone} disabled className="mt-1.5 opacity-60" />
   687→            <p className="text-xs text-muted-foreground mt-1">Le téléphone ne peut pas être modifié.</p>
   688→          </div>
   689→          <div>
   690→            <Label htmlFor="settings-email">Email (optionnel)</Label>
   691→            <Input id="settings-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} className="mt-1.5" placeholder="votre@email.com" />
   692→          </div>
   693→        </div>
   694→      </div>
   695→
   696→      <div className="border-t border-border pt-5">
   697→        <h3 className="text-sm font-semibold mb-4">Préférences</h3>
   698→        <div className="flex items-center justify-between bg-card rounded-xl p-4 border border-border">
   699→          <div className="flex items-center gap-3">
   700→            <Bell className="w-4 h-4 text-primary" />
   701→            <div>
   702→              <p className="text-sm font-medium">Notifications SMS</p>
   703→              <p className="text-xs text-muted-foreground">Recevoir un rappel avant votre créneau</p>
   704→            </div>
   705→          </div>
   706→          <Switch checked={notifications} onCheckedChange={setNotifications} />
   707→        </div>
   708→      </div>
   709→
   710→      <Button className="w-full" onClick={handleSave} disabled={saving}>
   711→        {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <><Save className="w-4 h-4 mr-1.5" /> Enregistrer</>}
   712→      </Button>
   713→      {saved && <p className="text-xs text-emerald-600 text-center">Paramètres sauvegardés.</p>}
   714→    </div>
   715→  );
