   176→   176→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
   177→   177→  const slots: TimeSlot[] = [];
   178→   178→  for (let h = 8; h <= 23; h++) {
   179→   179→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
   180→   180→    const available = availableHours ? !availableHours.has(timeStr) : true;
   181→   181→    slots.push({
   182→   182→      id: `slot-${h}`,
   183→   183→      time: timeStr,
   184→   184→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
   185→   185→      available,
   186→   186→      price: 25000,
   187→   187→    });
   188→   188→  }
   189→   189→  return slots;
   190→   190→}

   194→   194→   ═══════════════════════════════════════════ */
   195→   195→
   196→   196→function AnimatedCounter({ value, suffix, inView }: { value: number; suffix: string; inView: boolean }) {
   197→   197→  const [count, setCount] = useState(0);
   198→   198→  useEffect(() => {
   199→   199→    if (!inView) return;
   200→   200→    let start = 0;
   201→   201→    const step = Math.max(1, Math.floor(value / 125));
   202→   202→    const timer = setInterval(() => {
   203→   203→      start += step;
   204→   204→      if (start >= value) {
   205→   205→        setCount(value);
   206→   206→        clearInterval(timer);
   207→   207→      } else {
   208→   208→        setCount(start);
   209→   209→      }
   210→   210→    }, 16);
   211→   211→    return () => clearInterval(timer);
   212→   212→  }, [inView, value]);
   213→   213→
   214→   214→  return (
   215→   215→    <span>
   216→   216→      {count.toLocaleString('fr-FR')}
   217→   217→      {suffix}
   218→   218→    </span>
   219→   219→  );
   220→   220→}

   225→   225→
   226→   226→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   227→   227→  const ref = useRef<HTMLDivElement>(null);
   228→   228→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   229→   229→  return (
   230→   230→    <motion.section
   231→   231→      ref={ref}
   232→   232→      id={id}
   233→   233→      className={className}
   234→   234→      initial={{ opacity: 0, y: 30 }}
   235→   235→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   236→   236→      transition={{ duration: 0.6, ease: 'easeOut' }}
   237→   237→    >
   238→   238→      {children}
   239→   239→    </motion.section>
   240→   240→  );
   241→   241→}
