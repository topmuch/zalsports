    23→/* ═══════════════════════════════════════════
    24→   Types
    25→   ═══════════════════════════════════════════ */
    26→
    27→interface TimeSlot {
    28→  id: string;
    29→  time: string;
    30→  label: string;
    31→  available: boolean;
    32→  price: number;
    33→}
    34→
    35→const VENUE = {
    36→  pricePerHour: '25 000 FCFA',
    37→};
    38→
    39→type Step = 'select' | 'info' | 'payment' | 'confirm';
    40→
    41→const STEP_CONFIG: { key: Step; label: string; icon: typeof CalendarDays }[] = [
    42→  { key: 'select', label: 'Créneau', icon: CalendarDays },
    43→  { key: 'info', label: 'Informations', icon: UserCheck },
    44→  { key: 'payment', label: 'Paiement', icon: CreditCard },
    45→];
    46→
    47→function generateTimeSlots(availableHours?: Set<string>): TimeSlot[] {
    48→  const slots: TimeSlot[] = [];
    49→  for (let h = 8; h <= 23; h++) {
    50→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    51→    const available = availableHours ? !availableHours.has(timeStr) : true;
    52→    slots.push({
    53→      id: `slot-${h}`,
    54→      time: timeStr,
    55→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
    56→      available,
    57→      price: 25000,
    58→    });
    59→  }
    60→  return slots;
    61→}
    62→
    63→/* ═══════════════════════════════════════════
    64→   Slot Grid Button
    65→   ═══════════════════════════════════════════ */
    66→
    67→function SlotGridButton({ slot, selected, onClick }: { slot: TimeSlot; selected: boolean; onClick: () => void }) {
    68→  return (
    69→    <button
    70→      disabled={!slot.available}
    71→      onClick={onClick}
    72→      className={
    73→        `relative rounded-xl p-3 text-sm font-bold transition-all duration-200 ${
    74→          !slot.available
    75→            ? 'bg-red-50 border-2 border-red-300 text-red-400 cursor-not-allowed line-through'
    76→            : selected
    77→              ? 'bg-green-800 text-white border-2 border-green-800 shadow-lg shadow-green-800/30 scale-105'
    78→              : 'bg-green-50 border-2 border-green-700 text-green-900 hover:bg-green-100 hover:border-green-800'
    79→        }`
    80→      }
    81→    >
    82→      {slot.time}
    83→      {!slot.available && (
    84→        <span className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-red-400" />
    85→      )}
    86→      {selected && (
    87→        <Check className="absolute top-1.5 right-1.5 w-3.5 h-3.5 text-white" />
    88→      )}
    89→    </button>
    90→  );
    91→}

   156→    fetch(`/api/bookings/month-availability?month=${monthStr}`)
   157→      .then((res) => res.ok ? res.json() : null)

   159→        if (data?.fullyBooked) setFullyBookedDates(data.fullyBooked);

   172→          date: format(selectedDate, 'yyyy-MM-dd'),
   173→          timeSlot: selectedSlot.time,

   175→          customerPhone: phone,
   176→        }),

   182→        setStep('payment');
   183→      }

   185→      // silent
   186→    } finally {
   187→      setIsSubmitting(false);
   188→    }

   210→    } finally {

   222→            <button onClick={onBack} className="p-2 rounded-lg hover:bg-gray-100 transition-colors" aria-label="Retour">
   223→              <ArrowLeft className="w-5 h-5" />

   225→            <h1 className="text-lg font-bold">Réservation</h1>

   237→
   238→            <div className="bg-gray-50 rounded-2xl p-5 space-y-3 text-left">
   239→              <div className="flex justify-between text-sm">

   267→              Retour à l'accueil

   280→          <button onClick={onBack} className="p-2 rounded-lg hover:bg-gray-100 transition-colors" aria-label="Retour">
   281→            <ArrowLeft className="w-5 h-5" />
   282→          </button>
   283→          <h1 className="text-lg font-bold">Réservation</h1>
   284→          <div className="flex-1" />

   295→                    isActive ? 'bg-green-100 text-green-800' : isDone ? 'bg-green-50 text-green-600' : 'text-muted-foreground/50'
   296→                  }`}>
   297→                    {isDone ? <Check className="w-3.5 h-3.5" /> : <Icon className="w-3.5 h-3.5" />}
   298→                    <span>{s.label}</span>
   299→                  </div>
   300→                </div>
   301→              );
   302→            })}
   303→          </div>

   305→        {/* Mobile step indicator */}
   306→        <div className="sm:hidden border-t border-border/50 px-4 py-2">
   307→          <div className="flex items-center gap-2">
   308→            {STEP_CONFIG.map((s, i) => {
   309→              const Icon = s.icon;
   310→              const isActive = i === stepIndex;
   311→              const isDone = i < stepIndex;
   312→              return (
