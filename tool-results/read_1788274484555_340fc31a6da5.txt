    48→
    49→function isToday(date: Date): boolean {
    50→  const now = new Date();
    51→  return date.getFullYear() === now.getFullYear() &&
    52→    date.getMonth() === now.getMonth() &&
    53→    date.getDate() === now.getDate();
    54→}
    55→
    56→function generateTimeSlots(selectedDate?: Date, availableHours?: Set<string>): TimeSlot[] {
    57→  const slots: TimeSlot[] = [];
    58→  const now = new Date();
    59→  const currentHour = now.getHours();
    60→  const today = isToday(selectedDate || new Date());
    61→
    62→  for (let h = 8; h <= 23; h++) {
    63→    const timeStr = `${h.toString().padStart(2, '0')}:00`;
    64→    const booked = availableHours ? availableHours.has(timeStr) : false;
    65→    const past = today && h <= currentHour;
    66→    const available = !booked && !past;
    67→    slots.push({
    68→      id: `slot-${h}`,
    69→      time: timeStr,
    70→      label: `${timeStr} – ${(h + 1).toString().padStart(2, '0')}:00`,
    71→      available,
    72→      past,
    73→      price: 25000,
    74→    });
    75→  }
    76→  return slots;
    77→}

    82→
    83→function SlotGridButton({ slot, selected, onClick }: { slot: TimeSlot; selected: boolean; onClick: () => void }) {
    84→  const pastStyle = 'bg-gray-100 border-2 border-gray-200 text-gray-400 cursor-not-allowed';
    85→  const bookedStyle = 'bg-red-50 border-2 border-red-300 text-red-400 cursor-not-allowed line-through';
    86→  const selectedStyle = 'bg-green-800 text-white border-2 border-green-800 shadow-lg shadow-green-800/30 scale-105';
    87→  const availableStyle = 'bg-green-50 border-2 border-green-700 text-green-900 hover:bg-green-100 hover:border-green-800';
    88→
    89→  const style = slot.past ? pastStyle : !slot.available ? bookedStyle : selected ? selectedStyle : availableStyle;
    90→
    91→  return (
    92→    <button
    93→      disabled={!slot.available}
    94→      onClick={onClick}
    95→      className={`relative rounded-xl p-3 text-sm font-bold transition-all duration-200 ${style}`}
    96→    >
    97→      {slot.time}
    98→      {slot.past && (
    99→        <span className="absolute top-1.5 right-1.5 text-[9px] text-gray-400 font-medium">Passé</span>
   100→      )}
   101→      {!slot.available && !slot.past && (
   102→        <span className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-red-400" />
   103→      )}
   104→      {selected && (
   105→        <Check className="absolute top-1.5 right-1.5 w-3.5 h-3.5 text-white" />
   106→      )}
   107→    </button>
   108→  );
   109→}
